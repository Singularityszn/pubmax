  setup: { completed: false }, samples: [], privacyFailures: [], setupFailures: [] };

const browser = await chromium.launch(uiUxChromiumLaunchOptions(undefined));
const contexts: any[] = [];

const deadline = setTimeout(() => {

  report.setupFailures.push('HARD_DEADLINE_180_SECONDS');

const storageState = { cookies: [], origins: [{ origin, localStorage: [
  { name: 'pubmaxx:analytics-consent:v1', value: 'denied' },

async function context(width: number) {
  const result = await browser.newContext({ ...devices['Desktop Chrome'], baseURL: origin,
    viewport: { width, height: width === 390 ? 844 : 1000 },
    reducedMotion: 'reduce', serviceWorkers: 'block', storageState });
  result.setDefaultTimeout(15_000);
  contexts.push(result);
  return result;
}
function requireSetup(condition: unknown, code: string) {
  if (!condition) throw new Error(code);
}
async function checkedJson(response: any, stage: string) {
  report.setup[`${stage}Status`] = response.status();
  requireSetup(response.ok(), `${stage.toUpperCase()}_HTTP_NOT_OK`);
  try { return await response.json(); } catch { throw new Error(`${stage.toUpperCase()}_NON_JSON`); }
}

try {
  const owner = await context(1440);
  const slim = await checkedJson(await owner.request.get('/data/venues_slim.json', { timeout: 20_000 }), 'slim');
  // Real canonical row from the served dataset, not invented coordinates/names.
  const anchor = slim.rows?.find((row: any) => row.id === 'venue-11e0hkh'
    && row.filterHints?.canonical === true && typeof row.name === 'string');
  requireSetup(anchor, 'CANONICAL_CLAPHAM_ANCHOR_MISSING');
  const operationKey = randomUUID();
  const generated = await checkedJson(await owner.request.post('/api/plans/generate', {
    timeout: 40_000,
    data: { query: 'Beer in Clapham for 2, three pub stops', cityId: 'london', operationKey,
      context: { stopCount: 3 },
      anchor: { venueId: anchor.id, source: 'map-search', acceptedArea: null, startsAt: null } },
  }), 'generation');
  requireSetup(generated.anchored === true && generated.anchorVenueId === anchor.id
    && ['route', 'anchor-only'].includes(generated.outcome)
    && generated.operationKey === operationKey && typeof generated.groundingProof === 'string'
    && generated.stops?.length > 0 && generated.stops[0].venueId === anchor.id,
  'REAL_ANCHORED_GENERATION_NOT_ESTABLISHED');
  const privateTitle = `Privacy fixture ${randomUUID()}`;
  const guestName = `Private guest ${randomUUID().slice(0, 8)}`;
  const created = await checkedJson(await owner.request.post('/api/plans', {
    timeout: 25_000, headers: { 'idempotency-key': operationKey },
    data: { title: privateTitle, creatorName: 'Fixture host',
      startTime: new Date(Date.now() + 3 * 60 * 60 * 1000).toISOString(), cityId: 'london',
      stops: generated.stops, context: generated.inferredContext,
      groundingProof: generated.groundingProof,
      anchor: { venueId: anchor.id, source: generated.anchorSource, outcome: generated.outcome } },
  }), 'creation');
  const id = created.plan?.plan?.id;
  requireSetup(typeof id === 'string' && created.plan.plan.anchorVenueId === anchor.id
    && created.plan.stops[0]?.venueId === anchor.id, 'CREATED_PLAN_NOT_ACTUALLY_ANCHORED');
  const hostState = await checkedJson(await owner.request.get(`/api/plans/${id}`), 'hostRead');
  requireSetup(hostState.visibility === undefined && hostState.plan?.anchorVenueId === anchor.id
    && hostState.stops?.length === generated.stops.length
    && typeof hostState.inviteToken === 'string', 'HOST_CAPABILITY_READ_NOT_ESTABLISHED');
  const guest = await context(390);
  await checkedJson(await guest.request.post(`/api/plans/${id}/join`, {
    headers: { 'idempotency-key': randomUUID() },
    data: { name: guestName, inviteToken: hostState.inviteToken },
  }), 'join');
  const guestState = await checkedJson(await guest.request.get(`/api/plans/${id}`), 'guestRead');
  requireSetup(guestState.visibility === undefined && guestState.plan?.anchorVenueId === anchor.id
    && guestState.stops?.length === generated.stops.length
    && guestState.crew?.some((row: any) => row.name === guestName), 'GUEST_CAPABILITY_READ_NOT_ESTABLISHED');
  const ownerPage = await owner.newPage();
  requireSetup((await ownerPage.goto(`/plan/${id}`, { waitUntil: 'domcontentloaded', timeout: 30_000 }))?.status() === 200,
    'HOST_DOCUMENT_NOT_OK');
  // No manual token injection: real HttpOnly API cookie drives client upgrade.
  await expect(ownerPage.locator('.planRoute').getByText(anchor.name, { exact: true }).first())
    .toBeVisible({ timeout: 25_000 });
  report.setup = { ...report.setup, completed: true, anchorVenueId: anchor.id,
    anchorVenueName: anchor.name, actualOutcome: generated.outcome,
    actualStopCount: generated.stops.length, hostMemberRead: true,
    joinedGuestMemberRead: true, nativeHostRouteVisible: true };
  // Never persist create, generation, member JSON, headers or storageState:
  // these carry signed proof/member/invite capability values.
  const forbidden = [...new Set([anchor.id, privateTitle, guestName,
    ...hostState.stops.flatMap((row: any) => [row.venueId, row.venueName])])]
    .filter((value): value is string => typeof value === 'string' && value.length > 0);
  for (const width of [390, 1440]) {
    const anonymous = await context(width);
    const page = await anonymous.newPage();
    const sample: any = { width, completed: false, responses: [], failures: [] };
    report.samples.push(sample);
    const pending: Promise<void>[] = [];

        containsAnchorId: body.includes(anchor.id), forbiddenTokenCount: leaked.length,
