// UNRUN draft. Parent owns runtime. Three fresh contexts, original oracle retained.
// No app/map writes except deliberately dispatching GL fault events or using WEBGL_lose_context.
import { mkdirSync, writeFileSync } from 'node:fs';
import { chromium, devices, expect } from '/Users/karanmanoharan/.codex/worktrees/v0-integration/pubmaxx/node_modules/@playwright/test/index.mjs';
import { uiUxChromiumLaunchOptions } from '/Users/karanmanoharan/.codex/worktrees/v0-integration/pubmaxx/scripts/lib/uiUxBattleTestBrowser.mjs';

const origin = new URL(process.argv[2] ?? 'http://localhost:3351').origin;
if (!['localhost', '127.0.0.1'].includes(new URL(origin).hostname)) throw new Error('Loopback only');
const output = process.argv[3] ?? '/tmp/pubmaxx-r26-webgl-native-diagnostic';
mkdirSync(output, { recursive: true });
const report: any = { completed: false, origin, originalSource: 'e2e/map-webgl-recovery.spec.ts',
  limits: { totalMs: 150000, armMs: 45000, navigationMs: 30000, visibleMs: 25000, listeningMs: 30000, restoredMs: 5000, paintedMs: 15000 },
  limitations: 'Diagnostic observers add overhead. LISTENING is original wiring boundary, not painted readiness. Public painted probe reads actual placed pub pins/clusters; opening clusters are pub marks, not individual venue-sheet proof. No physical-device claim. Requires parent-started fresh keyless production build; script does not start or verify server configuration.',
  arms: [], errors: [] };
const flush = () => writeFileSync(`${output}/report.json`, JSON.stringify(report, null, 2) + '\n');
flush();
let browser: any;
let context: any;
let globalExpired = false;
const globalTimer = setTimeout(() => {
  globalExpired = true;
  report.errors.push('145s cleanup deadline; 150s hard exit cap'); flush();
  // Exit remains failure even if a stuck launch/close would otherwise keep Node alive.
  void browser?.close().catch(() => {});
  setTimeout(() => process.exit(1), 5000).unref();
}, 145000);
const snapshot = `(() => {
  const surface = document.querySelector('.maplibreMap');
  const canvas = surface?.querySelector('canvas');
  let camera = null, points = [], probeError = null;
  try { camera = window.__pubmaxMapCamera?.read() ?? null; points = window.__pubmaxPaintedMapTapPoints?.() ?? []; }
  catch (error) { probeError = String(error); }
  return { at: performance.now(), recovery: surface?.getAttribute('data-webgl-recovery'),
    canvasMounted: Boolean(canvas?.isConnected), canvasRect: canvas ? (() => { const r = canvas.getBoundingClientRect(); return { x:r.x,y:r.y,width:r.width,height:r.height }; })() : null,
    canvasCount: document.querySelectorAll('.maplibreMap canvas').length,
    fallbackCount: document.querySelectorAll('.mapFallback').length,
    softRetry: Array.from(document.querySelectorAll('[role="status"], [role="alert"]')).map(e => e.textContent),
    camera, points, probeError,
    marks: performance.getEntriesByType('mark').filter(e => e.name.startsWith('pubmax:')).map(e => ({ name:e.name,at:e.startTime })) };
})()`;

try {
  browser = await chromium.launch(uiUxChromiumLaunchOptions(process.env.UI_UX_BROWSER_CHANNEL));
  for (const mode of ['original-synthetic-loss', 'synthetic-loss-and-restore', 'native-loss-and-restore']) {
    if (globalExpired) break;
    const arm: any = { mode, completed: false, passed: false, console: [], pageErrors: [], journal: [], network: [], snapshots: [], errors: [] };
    report.arms.push(arm); flush();
    let armTimer: any;
    let armExpired = false;
    let page: any;
    try {
      context = await browser.newContext({ ...devices['Desktop Chrome'], viewport: { width:390, height:844 },
        baseURL: origin, serviceWorkers: 'block', permissions: [],
        storageState: { cookies: [], origins: [{ origin, localStorage: [
          { name:'pubmaxx:analytics-consent:v1', value:'denied' },
          { name:'pubmax:map-first-visit-arrival:v1', value:'dismissed' },
        ] }] } });
      const armContext = context;
      armTimer = setTimeout(() => { armExpired = true; arm.errors.push('45s arm deadline'); flush(); void armContext.close(); }, 45000);
      page = await context.newPage();
      page.setDefaultTimeout(5000);
      page.on('console', message => { arm.console.push({ at:Date.now(),type:message.type(),text:message.text(),location:message.location() }); flush(); });
      page.on('pageerror', error => { arm.pageErrors.push({ at:Date.now(),message:String(error),stack:error.stack }); flush(); });
      page.on('requestfailed', request => { arm.network.push({ at:Date.now(),kind:'failed',url:request.url(),failure:request.failure() }); flush(); });
      page.on('response', response => {
        const url = response.url();
        if (/style\.json|\/styles\/|\/api\/venues\/|venues_slim/.test(url)) {
          arm.network.push({ at:Date.now(),kind:'response',url,status:response.status() }); flush();
        }
      });
      // Binding journals arrive in Node independently of the fault/poll evaluation.
      await page.exposeBinding('__pubmaxR26Record', (_source: any, entry: any) => {
        if (arm.journal.length < 2000) arm.journal.push({ receivedAt:Date.now(),...entry });
        flush();
      });
      // Raw strings avoid tsx serialized-function __name helpers in browser realm.
      await page.addInitScript(`
        localStorage.setItem('pubmax-tour-v1-done','1');
        localStorage.setItem('pubmax_onboarding_dismissed','1');
        sessionStorage.setItem('pubmax_onboarding_dismissed','1');
        (() => {
          const emit = item => { void window.__pubmaxR26Record({ at:performance.now(),...item }).catch(() => {}); };
          const read = () => ${snapshot};
          let previous = null;
          const sample = () => {
            const s = read();
            const key = JSON.stringify({ recovery:s.recovery,canvasMounted:s.canvasMounted,canvasCount:s.canvasCount,fallbackCount:s.fallbackCount,points:s.points,probeError:s.probeError });
            if (key !== previous) { previous = key; emit({ kind:'surface',snapshot:s }); }
          };
          new MutationObserver(sample).observe(document, { subtree:true,childList:true,attributes:true,attributeFilter:['data-webgl-recovery'] });
          setInterval(sample,250);
          try { new PerformanceObserver(list => list.getEntries().filter(e => e.name.startsWith('pubmax:')).forEach(e => emit({kind:'mark',name:e.name,start:e.startTime}))).observe({type:'mark',buffered:true}); } catch {}
          for (const type of ['webglcontextlost','webglcontextrestored']) {
            document.addEventListener(type,event => {
              emit({kind:'gl-event-capture',type,trusted:event.isTrusted,defaultPrevented:event.defaultPrevented});
              setTimeout(() => emit({kind:'gl-event-after-dispatch',type,trusted:event.isTrusted,defaultPrevented:event.defaultPrevented,snapshot:read()}),0);
            },true);
          }
          window.addEventListener('pubmax:pin-reveal',event => emit({kind:'pin-reveal',detail:event.detail}));
        })();
      `);
      const response = await page.goto('/map', { timeout:30000 });
      expect(response?.status()).toBe(200);
      const canvas = page.locator('.maplibreMap canvas').first();
      await expect(canvas).toBeVisible({ timeout:25000 });
      await expect.poll(() => page.locator('.maplibreMap').getAttribute('data-webgl-recovery'), { timeout:30000 }).toBe('listening');
      arm.snapshots.push({ phase:'listening-before-fault',snapshot:await page.evaluate(snapshot) }); flush();

      if (mode === 'native-loss-and-restore') {
        arm.fault = await page.evaluate(`(() => {
          const canvas = document.querySelector('.maplibreMap canvas');
          const gl = canvas?.getContext('webgl2') || canvas?.getContext('webgl');
          const extension = gl?.getExtension('WEBGL_lose_context');
          if (!extension) return { ok:false,reason:'WEBGL_lose_context unavailable' };
          window.__pubmaxR26Fault = { gl,extension,canvas };
          extension.loseContext();
          return { ok:true,at:performance.now(),lost:gl.isContextLost() };
        })()`);
        expect(arm.fault.ok).toBe(true);
        await expect.poll(() => arm.journal.some((entry: any) => entry.kind === 'gl-event-after-dispatch' && entry.type === 'webglcontextlost' && entry.trusted), { timeout:5000 }).toBe(true);
        await expect.poll(() => arm.journal.some((entry: any) => entry.kind === 'gl-event-after-dispatch' && entry.type === 'webglcontextlost' && entry.defaultPrevented), { timeout:5000 }).toBe(true);
        arm.snapshots.push({ phase:'native-lost',snapshot:await page.evaluate(snapshot) });
        arm.restore = await page.evaluate(`(() => { const f=window.__pubmaxR26Fault; if(!f) return {ok:false}; f.extension.restoreContext(); return {ok:true,at:performance.now()}; })()`);
        expect(arm.restore.ok).toBe(true);
        await expect.poll(() => arm.journal.some((entry: any) => entry.kind === 'gl-event-after-dispatch' && entry.type === 'webglcontextrestored' && entry.trusted), { timeout:5000 }).toBe(true);
      } else {
        arm.fault = await page.evaluate(`(() => {
          const el=document.querySelector('.maplibreMap canvas');
          if(!el) return {ok:false,reason:'no-canvas'};
          const event=new Event('webglcontextlost',{cancelable:true,bubbles:true});
          el.dispatchEvent(event);
          const recovery=document.querySelector('.maplibreMap')?.getAttribute('data-webgl-recovery');
          ${mode === 'synthetic-loss-and-restore' ? "el.dispatchEvent(new Event('webglcontextrestored',{cancelable:true,bubbles:true}));" : ''}
          return {ok:true,at:performance.now(),defaultPrevented:event.defaultPrevented,recovery};
        })()`);
        expect(arm.fault.ok).toBe(true);
        expect(arm.fault.defaultPrevented).toBe(true);
        expect(arm.fault.recovery).toBe('recovering');
        if (mode === 'synthetic-loss-and-restore') {
          await expect.poll(() => page.locator('.maplibreMap').getAttribute('data-webgl-recovery'), { timeout:5000 }).toBe('restored');
        }
      }
      // Original first test's immediate mounted/no-fallback and 1200ms recovery oracle.
      await expect(canvas).toBeVisible();
      await expect(page.locator('.mapFallback')).toHaveCount(0);
      await page.waitForTimeout(1200);
      const after = await page.locator('.maplibreMap').getAttribute('data-webgl-recovery');
      expect(['restored','recovering','reinit','listening','soft-retry']).toContain(after);
      await expect(page.locator('.mapFallback')).toHaveCount(0);
      await expect(canvas).toBeVisible();
      arm.originalMountedOraclePassed = true;
      // Stronger live-pub rendering proof beside original oracle, without moving camera.
      await expect.poll(async () => {
        const s = await page.evaluate(snapshot);
        return s.fallbackCount === 0 && s.canvasMounted && !s.probeError && s.points.length > 0;
      }, { timeout:15000 }).toBe(true);
      arm.snapshots.push({ phase:'after-painted-proof',snapshot:await page.evaluate(snapshot) });
      arm.passed = true;
    } catch (error) {
      arm.errors.push(String(error));
    } finally {
      if (!armExpired && page && !page.isClosed()) {
        try { arm.snapshots.push({ phase:'final',snapshot:await page.evaluate(snapshot) }); } catch (error) { arm.errors.push(`final snapshot: ${String(error)}`); }
        try { await page.screenshot({ path:`${output}/${mode}.png`,timeout:3000 }); } catch (error) { arm.errors.push(`screenshot: ${String(error)}`); }
      }
      clearTimeout(armTimer);
      await context?.close().catch(() => {}); context = undefined;
      arm.completed = true; flush();
    }
  }
  report.completed = !globalExpired && report.arms.length === 3 && report.arms.every((arm: any) => arm.completed);
} catch (error) {
  report.errors.push(String(error));
} finally {
  await context?.close().catch(() => {});
  await browser?.close().catch(() => {});
  clearTimeout(globalTimer); flush();
}
// Any original or control failure remains EXIT1. Successful control never masks RED.
process.exitCode = report.completed && report.errors.length === 0 && report.arms.every((arm: any) => arm.passed && arm.errors.length === 0) ? 0 : 1;
