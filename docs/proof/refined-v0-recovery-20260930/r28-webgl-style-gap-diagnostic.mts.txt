const lossRecord=()=>report.journal.find((entry:any)=>entry.kind==='native-loss'&&entry.trusted);
const classify=()=>{

    sceneAlreadyFailed,exercised,reason:exercised?'Trusted native loss occurred after first style-load mark while no scene-built mark or completed scene failure existed.':
