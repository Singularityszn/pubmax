// UNRUN source draft. Requires explicit next coordinator runtime lease.
// Hold one real MapLibre response, prove real Inspector before release, then
// release unchanged bytes and prove actual canvas/pins. No performance verdict.
import { mkdirSync, writeFileSync, readFileSync } from 'node:fs';
import { chromium, devices, expect } from '/Users/karanmanoharan/.codex/worktrees/v0-integration/pubmaxx/node_modules/@playwright/test/index.mjs';
import { applyVitalsDevice, coolBrowser, clearSampleResidue, serveMapInitScript, SHIPPED_MAP_INIT_SCRIPT } from '/Users/karanmanoharan/.codex/worktrees/v0-integration/pubmaxx/e2e/helpers/webVitals.ts';
import { uiUxChromiumLaunchOptions } from '/Users/karanmanoharan/.codex/worktrees/v0-integration/pubmaxx/scripts/lib/uiUxBattleTestBrowser.mjs';

const origin = new URL(process.argv[2] ?? 'http://localhost:3351').origin;
if (!['localhost', '127.0.0.1'].includes(new URL(origin).hostname)) throw new Error('Loopback only');
const output = process.argv[3] ?? '/tmp/pubmaxx-r20-inspector-map-hold';
const mapPath = '/_next/static/chunks/21ctu9vnullj5.js';
const asset = '/Users/karanmanoharan/.codex/worktrees/v0-integration/pubmaxx/.next-e2e-recovery/static/chunks/21ctu9vnullj5.js';
if (!readFileSync(asset, 'utf8').includes('maplibregl')) throw new Error('R18 MapLibre ownership changed; learn actual emitted chunk before running');
mkdirSync(output, { recursive: true });
const report = { diagnosticOnly: true, processPid: process.pid, source: 'unchanged R18 build', mapPath, held: 0, released: false, errors: [], snapshots: [] };
const flush = () => writeFileSync(`${output}/report.json`, JSON.stringify(report, null, 2) + '\n');
flush();
const browser = await chromium.launch(uiUxChromiumLaunchOptions(process.env.UI_UX_BROWSER_CHANNEL));
let release;
let holdCeiling;
const barrier = new Promise(resolve => { release = resolve; });
const deadline = setTimeout(() => { report.errors.push('90s owned diagnostic deadline'); release(); void browser.close(); }, 90_000);
try {
  const context = await browser.newContext({ ...devices['Desktop Chrome'], baseURL: origin, serviceWorkers: 'block', storageState: { cookies: [], origins: [{ origin, localStorage: [{ name: 'pubmaxx:analytics-consent:v1', value: 'denied' }, { name: 'pubmax:map-first-visit-arrival:v1', value: 'dismissed' }] }] } });
  const page = await context.newPage();
  page.on('pageerror', error => report.errors.push(String(error)));
  await serveMapInitScript(page, SHIPPED_MAP_INIT_SCRIPT);
  await page.goto('/plan', { waitUntil: 'load' });
  await applyVitalsDevice(page, 'mobile');
  await coolBrowser(page);
  await clearSampleResidue(page, '/map?sel=venue-4xlgb0');
  await page.route(url => url.origin === origin && url.pathname === mapPath, async route => {
    report.held++;
    flush();
    const response = await route.fetch();
    await barrier;
    await route.fulfill({ response });
  });
  const navigationAt = Date.now();
  holdCeiling = setTimeout(() => { report.errors.push('16s pre-watchdog hold window expired'); report.released = true; release(); }, 16_000);
  await page.goto('/map?sel=venue-4xlgb0', { waitUntil: 'domcontentloaded' });
  await expect.poll(() => report.held, { timeout: 12_000 }).toBe(1);
  const inspector = page.locator('.venueInspector');
  // Within existing shell's 18s readiness watchdog. A timeout remains a failure;
  // it is not evidence of chunk causality without inspecting held state.
  await expect(inspector).toBeVisible({ timeout: 12_000 });
  await expect(page.getByRole('heading', { name: 'Princess Louise', exact: true })).toBeVisible();
  const price = inspector.locator('.trustPill[data-standing="estimate"]');
  await expect(price).toBeVisible();
  await expect(price).toContainText('est. £6.50');
  await expect(price).toContainText('Estimated');
  await expect(inspector.getByRole('tab', { name: 'Overview', exact: true })).toBeVisible();
  const before = await page.evaluate(() => ({ time: performance.now(), title: document.querySelector('.mobileSharedSheetHeader h2')?.textContent, inspectorTitle: document.querySelector('.venueInspector h3')?.textContent, price: document.querySelector('.venueInspector .trustPill[data-standing="estimate"]')?.textContent, inspector: document.querySelector('.venueInspector')?.getBoundingClientRect().toJSON(), mapCanvases: document.querySelectorAll('.maplibregl-canvas').length, marks: performance.getEntriesByType('mark').map(entry => entry.toJSON()), resources: performance.getEntriesByType('resource').map(entry => entry.toJSON()) }));
  report.snapshots.push({ phase: 'before-release', ...before });
  if (before.mapCanvases !== 0 || report.released || report.held !== 1) throw new Error('Map hold failed to isolate real dependency');
  await expect(page.locator('.mapFallback')).toHaveCount(0);
  if (Date.now() - navigationAt >= 16_000) throw new Error('Hold proof missed pre-watchdog boundary');
  await page.screenshot({ path: `${output}/before-release.png`, fullPage: false });
  report.released = true;
  clearTimeout(holdCeiling);
  release();
  clearTimeout(holdCeiling);
  flush();
  await expect(page.locator('.maplibregl-canvas')).toBeVisible({ timeout: 20_000 });
  await expect(page.getByRole('heading', { name: 'Princess Louise', exact: true })).toBeVisible();
  report.snapshots.push({ phase: 'named-inspector-after-release', ...await page.evaluate(() => ({ time: performance.now(), camera: window.__pubmaxMapCamera?.read(), marks: performance.getEntriesByType('mark').map(entry => entry.toJSON()) })) });
  try {
    await expect.poll(() => page.evaluate(() => typeof window.__pubmaxPaintedMapTapPoints === 'function' ? window.__pubmaxPaintedMapTapPoints().length : 0), { timeout: 25_000 }).toBeGreaterThan(0);
    report.initialPinOracle = 'passed';
  } catch (error) {
    report.initialPinOracle = 'failed';
    report.errors.push(String(error));
    process.exitCode = 1;
    report.snapshots.push({ phase: 'failed-original-oracle', ...await page.evaluate(() => ({ time: performance.now(), camera: window.__pubmaxMapCamera?.read(), points: window.__pubmaxPaintedMapTapPoints?.(), modal: document.querySelector('.mobileSharedSheet')?.getAttribute('aria-modal'), fallback: document.querySelector('.mapFallback')?.textContent, hitStack: document.elementsFromPoint(195, 260).map(element => ({ tag: element.tagName, className: element.className, pointerEvents: getComputedStyle(element).pointerEvents })), marks: performance.getEntriesByType('mark').map(entry => entry.toJSON()) })) });
    await page.screenshot({ path: `${output}/failed-original-oracle.png`, fullPage: false });
    flush();
  }
  // A separate native-close control diagnoses the original oracle failure;
  // it never replaces or converts that failure into an overall passing run.
  await page.keyboard.press('Escape');
  await expect(inspector).toHaveCount(0);
  await expect.poll(() => page.evaluate(() => typeof window.__pubmaxPaintedMapTapPoints === 'function' ? window.__pubmaxPaintedMapTapPoints().length : 0), { timeout: 25_000 }).toBeGreaterThan(0);
  report.afterNativeCloseControl = 'passed';
  report.snapshots.push({ phase: 'after-release', ...await page.evaluate(() => ({ time: performance.now(), camera: window.__pubmaxMapCamera?.read(), points: window.__pubmaxPaintedMapTapPoints?.(), marks: performance.getEntriesByType('mark').map(entry => entry.toJSON()) })) });
  await page.screenshot({ path: `${output}/after-release.png`, fullPage: false });
  if (report.errors.length && report.initialPinOracle !== 'failed') throw new Error('Page errors occurred');
  report.completed = true;
  await context.close();
} catch (error) {
  report.errors.push(String(error));
  process.exitCode = 1;
} finally {
  release();
  clearTimeout(holdCeiling);
  clearTimeout(deadline);
  await browser.close();
  flush();
}
